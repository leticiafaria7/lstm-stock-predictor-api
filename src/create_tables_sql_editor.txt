-- =====================================================================
-- historical_tickers
-- =====================================================================

CREATE TABLE IF NOT EXISTS historical_tickers (

    id BIGSERIAL PRIMARY KEY,

    "Date" DATE NOT NULL,
    "Ticker" VARCHAR(20) NOT NULL,

    "Open" DOUBLE PRECISION,
    "High" DOUBLE PRECISION,
    "Low" DOUBLE PRECISION,
    "Close" DOUBLE PRECISION,
    "Volume" DOUBLE PRECISION,

    CONSTRAINT uq_historical_tickers
        UNIQUE ("Date","Ticker")
);

-- =====================================================================
-- historical_assets
-- =====================================================================

CREATE TABLE IF NOT EXISTS historical_assets (

    "Date" DATE PRIMARY KEY,

    "Close_Ibovespa" DOUBLE PRECISION,
    "Close_SP_500" DOUBLE PRECISION,
    "Close_Dolar" DOUBLE PRECISION
);

-- =====================================================================
-- historical_indexes
-- =====================================================================

CREATE TABLE IF NOT EXISTS historical_indexes (

    "Date" DATE PRIMARY KEY,

    "Selic" DOUBLE PRECISION,
    "IPCA" DOUBLE PRECISION
);

-- =====================================================================
-- grant access and disable security to write
-- =====================================================================

GRANT USAGE ON SCHEMA public TO anon;

GRANT SELECT, INSERT, UPDATE, DELETE
ON TABLE historical_tickers
TO anon;

GRANT SELECT, INSERT, UPDATE, DELETE
ON TABLE historical_assets
TO anon;

GRANT SELECT, INSERT, UPDATE, DELETE
ON TABLE historical_indexes
TO anon;

GRANT USAGE, SELECT
ON SEQUENCE historical_tickers_id_seq
TO anon;

GRANT USAGE, SELECT
ON SEQUENCE historical_tickers_id_seq
TO authenticated;

ALTER TABLE historical_tickers DISABLE ROW LEVEL SECURITY;
ALTER TABLE historical_assets DISABLE ROW LEVEL SECURITY;
ALTER TABLE historical_indexes DISABLE ROW LEVEL SECURITY;

-- =====================================================================
-- create table predictions
-- =====================================================================

CREATE TABLE IF NOT EXISTS predictions (

    "Date" DATE NOT NULL,
    "Ticker" VARCHAR(20) NOT NULL,

    "Close" DOUBLE PRECISION NOT NULL,
    "Predict_D1" DOUBLE PRECISION NOT NULL,

    PRIMARY KEY ("Date","Ticker")
);

GRANT SELECT, INSERT, UPDATE, DELETE
ON TABLE predictions
TO anon;

ALTER TABLE predictions
DISABLE ROW LEVEL SECURITY;

-- =====================================================================
-- create table logs github actions
-- =====================================================================

create table public.workflow_logs (

    id bigint generated always as identity primary key,

    created_at timestamptz not null default now(),

    workflow text not null,
    status text not null,
    message text,

    duration_seconds numeric(10,2),

    rows_historical_tickers integer not null default 0,
    rows_historical_assets integer not null default 0,
    rows_historical_indexes integer not null default 0,
    rows_predictions integer not null default 0

);

grant select, insert on public.workflow_logs to anon;
grant usage, select on sequence public.workflow_logs_id_seq to anon;

alter table public.workflow_logs disable row level security;

-- =====================================================================
-- dashboard_cache
--
-- Tabela "single row" (id sempre = 1) que guarda todos os objetos já
-- prontos para renderizar a home (sidebar, cards, métricas e gráficos
-- em JSON). É atualizada 1x por dia no sync.py, evitando que a API
-- precise carregar os modelos LSTM e reprocessar as tabelas históricas
-- a cada request (o que estava estourando o limite de memória do
-- plano gratuito do Render).
-- =====================================================================

CREATE TABLE IF NOT EXISTS dashboard_cache (

    id SMALLINT PRIMARY KEY DEFAULT 1,

    updated_at TIMESTAMPTZ NOT NULL DEFAULT now(),

    sidebar JSONB NOT NULL,
    companies JSONB NOT NULL,
    features JSONB NOT NULL,
    endpoints JSONB NOT NULL,
    tickers JSONB NOT NULL,
    default_ticker TEXT NOT NULL,

    metrics JSONB NOT NULL,
    charts JSONB NOT NULL,

    CONSTRAINT single_row CHECK (id = 1)
);

GRANT SELECT, INSERT, UPDATE
ON TABLE dashboard_cache
TO anon;

ALTER TABLE dashboard_cache DISABLE ROW LEVEL SECURITY;
